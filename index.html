Thonny IDE 또는 VSCode MicroPico 확장 기능에서 ESP32 마이크로컨트롤러에 업로드할 수 있는 MicroPython 코드입니다.
네오픽셀 LED(GPIO 14번 핀에 연결, 4구)가 실시간으로 웹 앱으로부터 수신한 문자열 '1', '2', 'abc', '4'에 반응하여 색상을 제어합니다.

MicroPython - main.py
# ESP32 MicroPython BLE UART LED Control Example import bluetooth from machine import Pin import neopixel import time # 네오픽셀 설정 (GPIO 14, 4구) NUM_LEDS = 4 np = neopixel.NeoPixel(Pin(14), NUM_LEDS) # BLE UART UUIDs _UART_UUID = bluetooth.UUID("6e400001-b5a3-f393-e0a9-e50e24dcca9e") _UART_TX = bluetooth.UUID("6e400003-b5a3-f393-e0a9-e50e24dcca9e") _UART_RX = bluetooth.UUID("6e400002-b5a3-f393-e0a9-e50e24dcca9e") class BLEUART: def __init__(self, name="ESP_BLE_CTRL"): self._ble = bluetooth.BLE() self._ble.active(True) self._ble.irq(self._irq) ((self._tx, self._rx),) = self._ble.gatts_register_services( ((_UART_UUID, ((_UART_TX, bluetooth.FLAG_NOTIFY), (_UART_RX, bluetooth.FLAG_WRITE)),),) ) self._connections = set() self._write_callback = None self._payload = self._ad_payload(name=name, services=[_UART_UUID]) self._advertise() def _irq(self, event, data): if event == 1: # _IRQ_CENTRAL_CONNECT conn_handle, _, _ = data self._connections.add(conn_handle) print("Central connected") elif event == 2: # _IRQ_CENTRAL_DISCONNECT conn_handle, _, _ = data if conn_handle in self._connections: self._connections.remove(conn_handle) print("Central disconnected") self._advertise() elif event == 3: # _IRQ_GATTS_WRITE conn_handle, value_handle = data if value_handle == self._rx: data_bytes = self._ble.gatts_read(self._rx) if self._write_callback: self._write_callback(data_bytes) def on_write(self, callback): self._write_callback = callback def send(self, data): for conn_handle in self._connections: self._ble.gatts_notify(conn_handle, self._tx, data) def _advertise(self, interval_us=500000): self._ble.gap_advertise(interval_us, adv_data=self._payload) print("Advertising...") def _ad_payload(self, name=None, services=None): payload = bytearray() def _append(adv_type, value): nonlocal payload payload.append(len(value) + 1) payload.append(adv_type) payload.extend(value) # Flag (General discoverable mode, BR/EDR not supported) _append(0x01, b"\x06") if name: _append(0x09, name.encode()) if services: for uuid in services: b = bytes(uuid) if len(b) == 2: _append(0x03, b) elif len(b) == 16: _append(0x07, b) # 128-bit UUIDs return payload # LED 제어 로직 def handle_rx(data_bytes): # 디코딩 및 공백 제거 try: msg = data_bytes.decode('utf-8').strip() except Exception: return print("Received:", msg) # 받은 데이터에 따른 네오픽셀 LED 제어 # '1': 빨간색, '2': 초록색, 'abc': 파란색, '4': 노란색 if msg == '1': color_all((255, 0, 0)) # 빨간색 elif msg == '2': color_all((0, 255, 0)) # 초록색 elif msg == 'abc': color_all((0, 0, 255)) # 파란색 elif msg == '4': color_all((255, 255, 0)) # 노란색 else: # 기타 매핑값은 보라색으로 표시 color_all((128, 0, 128)) def color_all(color): for i in range(NUM_LEDS): np[i] = color np.write() # BLE 및 네오픽셀 초기화 print("NeoPixel self-test...") color_all((255, 0, 0)) # 빨간색 time.sleep(0.3) color_all((0, 255, 0)) # 초록색 time.sleep(0.3) color_all((0, 0, 255)) # 파란색 time.sleep(0.3) color_all((0, 0, 0)) # 꺼짐 print("Self-test completed.") uart = BLEUART(name="ESP_NEO_CTRL") uart.on_write(handle_rx) print("BLE UART Ready. Waiting for connection...") try: while True: time.sleep(1) except KeyboardInterrupt: color_all((0, 0, 0)) print("Stopped")
